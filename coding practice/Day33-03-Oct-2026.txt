# Ansible - Install Apache and Start Service
- name: Install and Start Apache
  hosts: all
  become: yes
  tasks:
    - name: Install Apache
      yum:
        name: httpd
        state: present

    - name: Start apache
      service: 
      name: httpd
      state: started
      enabled: yes

#Kubernetes - Create a ReplicaSet with 3 Replicas
apiVersion: apps/v1
kind: ReplicaSet
metadata:
  name: pathnex-replicaset
spec:
  replicas: 3
  selector:
    matchLables:
      app: pathnex-app
    template:
      metadata:
        lables:
          app" pathnex-app
        spec:
          conatiners:
            - name: app
              image: nginx
              ports:
                - containerPort: 80
