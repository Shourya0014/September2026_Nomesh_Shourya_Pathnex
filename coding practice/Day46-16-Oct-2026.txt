# Day 46 — Ansible Roles, DynamoDB, StatefulSet, Service Check
## Ansible — Create Role Structure

mkdir -p ansible/roles/pathnex/{tasks,handlers,templates,vars}

cat > ansible/roles/pathnex/tasks/main.yml <<'EOF'
---
- name: Install httpd
  yum:
    name: httpd
    state: present

- name: Deploy index page from template
  template:
    src: index.html.j2
    dest: /var/www/html/index.html
    mode: "0644"
  notify: Restart httpd

- name: Start and enable httpd
  service:
    name: httpd
    state: started
    enabled: yes
EOF

cat > ansible/roles/pathnex/handlers/main.yml <<'EOF'
---
- name: Restart httpd
  service:
    name: httpd
    state: restarted
EOF

cat > ansible/roles/pathnex/templates/index.html.j2 <<'EOF'
<!DOCTYPE html>
<html>
<head>
  <title>{{ institute_name }}</title>
</head>
<body>
  <h1>Welcome to {{ institute_name }}</h1>
  <p>Deployed using the Ansible role: pathnex</p>
</body>
</html>
EOF

cat > ansible/roles/pathnex/vars/main.yml <<'EOF'
---
institute_name: "Pathnex"
EOF

cat > ansible/site.yml <<'EOF'
---
- name: Apply pathnex role
  hosts: web
  become: true
  roles:
    - pathnex
EOF

cat > ansible/inventory.ini <<'EOF'
[web]
EC2_PUBLIC_IP ansible_user=ec2-user ansible_ssh_private_key_file=~/.ssh/my-key.pem
EOF

# Shell Script — Check Service Running
#!/bin/bash
SERVICE="sshd"

if systemctl is-active --quiet $SERVICE; then
  echo "$SERVICE is running"
else
  echo "$SERVICE is NOT running"
fi


# Artifacts Promotion
# Jenkins Pipeline — Promote Artifacts

pipeline {
    agent any
    environment {
        INSTITUTE_NAME = "Pathnex"
    }
    stages {
        stage('Build') {
            steps {
                sh 'mvn clean package'
            }
        }
        stage('Promote') {
            steps {
                input message: "Promote artifacts to staging?"
                sh 'echo "Promoting artifacts for $INSTITUTE_NAME"'
            }
        }
    }
}

# GitLab CI — Artifacts Promotion
stages:
  - build
  - promote

build:
  stage: build
  image: maven:3.8.1-jdk-17
  script:
    - mvn clean package
  artifacts:
    paths:
      - target/*.jar
    expire_in: 1 week

promote:
  stage: promote
  image: alpine:latest
  script:
    - echo "Promoting artifacts for Pathnex"
  when: manual
