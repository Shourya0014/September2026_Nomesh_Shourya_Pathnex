# Ansible:
- hosts: all
  become: yes
  tasks:
    - name: Create admin user
      user:
        name: pathnex-admin
        shell: /bin/bash
        state: present
    - name: Add SSH key
      authorized_key:
        user: pathnex-admin
        key: "{{ lookup('file', '~/.ssh/id_rsa.pub') }}"

# Shell — monitor.sh
#!/bin/bash
echo "1) IP  2) Disk  3) Memory"
read -p "Choose: " n
case $n in
  1) hostname -I ;;
  2) df -h ;;
  3) free -h ;;
  *) echo "Invalid choice" ;;
esac

# Jenkinsfile
pipeline {
  agent any
  environment {
    INSTITUTE_NAME = 'Pathnex'
    TEAM = 'DevOps'
    ENV = 'prod'
    PROJECT = 'Pathnex-Training'
  }
  stages {
    stage('Infrastructure') {
      steps {
        sh 'cd terraform && terraform init && terraform apply -auto-approve'
      }
    }
    stage('App Deployment') {
      steps {
        sh 'ansible-playbook user.yml'
        sh 'kubectl apply -f k8s.yml'
      }
    }
    stage('Monitoring Checks') {
      steps {
        sh 'bash monitor.sh'
      }
    }
  }
}
